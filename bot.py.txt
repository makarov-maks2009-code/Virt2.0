import asyncio
import logging
import os
import random
from aiogram import Bot, Dispatcher, F
from aiogram.types import Message, CallbackQuery
from aiogram.filters import Command
from aiogram.utils.keyboard import InlineKeyboardBuilder
from aiogram.fsm.context import FSMContext
from aiogram.fsm.state import State, StatesGroup
from aiogram.fsm.storage.memory import MemoryStorage

logging.basicConfig(level=logging.INFO)

BOT_TOKEN = os.getenv("BOT_TOKEN")
if not BOT_TOKEN:
    raise ValueError("BOT_TOKEN не найден! Добавь его в Variables на Railway.")

bot = Bot(token=BOT_TOKEN)
dp = Dispatcher(storage=MemoryStorage())

# === Данные ===
GENDERS = {
    "male": "Мужской",
    "female": "Женский",
    "femboy": "Фембой",
    "shemale": "Трап/Шимейл"
}

ROLES = {
    "dom": "Доминант",
    "sub": "Сабмиссив",
    "switch": "Свитч"
}

FETISHES = {
    "humiliation": "Чмор",
    "blackmail": "Шантаж",
    "milking": "Дойка",
    "orgasm_control": "Контроль оргазма",
    "foot": "Фут-фетиш",
    "boot": "Бут-фетиш",
    "parents_hum": "Чмор родителей",
    "spit": "Плевки",
    "toilet": "Туалетная тема",
    "facesitting": "Фейсситинг",
    "body_hum": "Чмор тела"
}

class VirtStates(StatesGroup):
    choosing_user_gender = State()
    choosing_bot_gender = State()
    choosing_user_role = State()
    choosing_bot_role = State()
    choosing_fetishes = State()
    in_roleplay = State()

# === Клавиатуры ===
def gender_kb(prefix: str):
    builder = InlineKeyboardBuilder()
    for key, name in GENDERS.items():
        builder.button(text=name, callback_data=f"{prefix}:{key}")
    builder.adjust(2)
    return builder.as_markup()

def role_kb(prefix: str):
    builder = InlineKeyboardBuilder()
    for key, name in ROLES.items():
        builder.button(text=name, callback_data=f"{prefix}:{key}")
    builder.adjust(1)
    return builder.as_markup()

def fetish_kb(selected: list):
    builder = InlineKeyboardBuilder()
    for key, name in FETISHES.items():
        mark = "✅ " if key in selected else ""
        builder.button(text=f"{mark}{name}", callback_data=f"fetish:{key}")
    builder.button(text="➡️ Готово", callback_data="fetish:done")
    builder.adjust(2)
    return builder.as_markup()

# === Старт ===
@dp.message(Command("start"))
async def start(message: Message, state: FSMContext):
    await state.clear()
    await state.set_state(VirtStates.choosing_user_gender)
    await message.answer(
        "Выбери <b>свой пол</b>:",
        reply_markup=gender_kb("user_gender"),
        parse_mode="HTML"
    )

# === Выбор пола пользователя ===
@dp.callback_query(F.data.startswith("user_gender:"), VirtStates.choosing_user_gender)
async def user_gender(callback: CallbackQuery, state: FSMContext):
    gender = callback.data.split(":")[1]
    await state.update_data(user_gender=gender)
    await state.set_state(VirtStates.choosing_bot_gender)
    await callback.message.edit_text(
        "Теперь выбери <b>пол бота</b>:",
        reply_markup=gender_kb("bot_gender"),
        parse_mode="HTML"
    )
    await callback.answer()

# === Выбор пола бота ===
@dp.callback_query(F.data.startswith("bot_gender:"), VirtStates.choosing_bot_gender)
async def bot_gender(callback: CallbackQuery, state: FSMContext):
    gender = callback.data.split(":")[1]
    await state.update_data(bot_gender=gender)

        "Выбери <b>свою роль</b>:",
        reply_markup=role_kb("user_role"),
        parse_mode="HTML"

    await callback.answer()

# === Выбор роли пользователя ===
@dp.callback_query(F.data.startswith("user_role:"), VirtStates.choosing_user_role)
async def user_role(callback: CallbackQuery, state: FSMContext):
    role = callback.data.split(":")[1]
    await state.update_data(user_role=role)
    await state.set_state(VirtStates.choosing_bot_role)
    await callback.message.edit_text(
        "Выбери <b>роль бота</b>:",

# === Выбор роли бота ===
@dp.callback_query(F.data.startswith("bot_role:"), VirtStates.choosing_bot_role)
async def bot_role(callback: CallbackQuery, state: FSMContext):
    role = callback.data.split(":")[1]
    await state.update_data(bot_role=role, selected_fetishes=[])
    await state.set_state(VirtStates.choosing_fetishes)
    await callback.message.edit_text(
        "Выбери фетиши (можно несколько).\nКогда закончишь — нажми «Готово»:",
        reply_markup=fetish_kb([])
    )
    await callback.answer()

# === Выбор фетишей ===
@dp.callback_query(F.data.startswith("fetish:"), VirtStates.choosing_fetishes)
async def choose_fetish(callback: CallbackQuery, state: FSMContext):
    data = await state.get_data()
    selected = data.get("selected_fetishes", [])
    fetish = callback.data.split(":")[1]

    if fetish == "done":
        if not selected:
            await callback.answer("Выбери хотя бы один фетиш!", show_alert=True)
            return
        await state.set_state(VirtStates.in_roleplay)
        await start_roleplay(callback.message, state)
        await callback.answer()
        return

    if fetish in selected:
        selected.remove(fetish)
    else:
        selected.append(fetish)

    await state.update_data(selected_fetishes=selected)
    await callback.message.edit_reply_markup(reply_markup=fetish_kb(selected))
    await callback.answer()

# === Старт ролевой ===
async def start_roleplay(message: Message, state: FSMContext):
    data = await state.get_data()
    
    user_g = GENDERS[data["user_gender"]]
    bot_g = GENDERS[data["bot_gender"]]
    user_r = ROLES[data["user_role"]]
    bot_r = ROLES[data["bot_role"]]
    fets = [FETISHES[f] for f in data["selected_fetishes"]]

    text = (
        f"<b>Ролевая начата</b>\n\n"
        f"Ты: {user_g}, {user_r}\n"
        f"Бот: {bot_g}, {bot_r}\n"
        f"Фетиши: {', '.join(fets)}\n\n"
        f"Пиши действия и слова. Я буду отвечать в выбранном стиле.\n\n"
        f"Команды:\n"
        f"/stop — остановить\n"
        f"/status — посмотреть настройки"
    )
    await message.edit_text(text, parse_mode="HTML")

# === Ролевая ===
@dp.message(VirtStates.in_roleplay)
async def roleplay(message: Message, state: FSMContext):
    if message.text and message.text.startswith("/"):
        return

    data = await state.get_data()
    reply = generate_response(data, message.text or "")
    await message.answer(reply)

@dp.message(Command("status"), VirtStates.in_roleplay)
async def status(message: Message, state: FSMContext):
    data = await state.get_data()
    fets = [FETISHES[f] for f in data["selected_fetishes"]]
    text = (
        f"Ты: {GENDERS[data['user_gender']]}, {ROLES[data['user_role']]}\n"
        f"Бот: {GENDERS[data['bot_gender']]}, {ROLES[data['bot_role']]}\n"
        f"Фетиши: {', '.join(fets)}"
    )
    await message.answer(text)

@dp.message(Command("stop"))
async def stop(message: Message, state: FSMContext):
    await state.clear()
    await message.answer("Ролевая остановлена.\nНапиши /start чтобы начать заново.")
